2. Graph Population Pipeline
Step 1: Extract Nodes & Edges
python
Copy
# In your data processing Repl
def build_graph(documents):
    for doc in documents:
        # Add document node
        g.add_node(
            id=f"doc_{doc['id']}",
            type="Document",
            props={"title": doc["title"]}
        )
        
        # Add entity edges
        for entity in doc["entities"]:
            g.add_node(
                id=f"ent_{entity['id']}",
                type="Entity",
                props={"name": entity["text"]}
            )
            g.add_edge(
                f"doc_{doc['id']}",
                f"ent_{entity['id']}",
                type="CONTAINS"
            )

    for i, j in np.argwhere(sim_matrix > threshold):
        if i != j:
            g.add_edge(
                f"doc_{i}",
                f"doc_{j}",
                type="RELATED",
                weight=float(sim_matrix[i][j])
            )
3. Graph Visualization with D3.js
Frontend Repl Setup
javascript
Copy
// In a React Repl (using vite)
import * as d3 from "d3";
import { ForceGraph } from "react-force-graph";

function GraphVisualization() {
  const [graphData, setGraphData] = useState({ nodes: [], links: [] });

  useEffect(() => {
    fetch("/api/graph")
      .then(res => res.json())
      .then(data => setGraphData(data));
  }, []);

  return (
    <ForceGraph
      graphData={graphData}
      nodeLabel="id"
      nodeAutoColorBy="type"
      linkWidth={link => link.weight * 2}
      onNodeClick={node => window.open(`/content/${node.id}`)}
    />
  );
}

Time Decay Implementation
python
Copy
# When querying relationships
def get_recent_connections(node_id, months=3):
    decay_factor = 0.9 ** months
    return [
        (edge.target, edge.weight * decay_factor)
        for edge in g.edges
        if edge.source == node_id 
        and edge.created_at > now() - timedelta(days=30*months)
    ]
